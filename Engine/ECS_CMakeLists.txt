cmake_minimum_required(VERSION 3.15)

# ECS Example executable (header-only, so just needs to compile the example)
add_executable(ECSExample src/ECS/ECSExample.cpp)

target_include_directories(ECSExample PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(ECSExample PRIVATE fmt)

# Set C++ standard
set_target_properties(ECSExample PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
)

install(TARGETS ECSExample RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
